/**
 * ✅ Визуализация на ВХОД → апартаменти
 * - показва всички апартаменти във входа
 * - hover показва статус + примерни детайли
 * - demo логика: paidOnTime / overdue / unpaid
 */
function EntranceApartmentsDemo() {
  const [entrance, setEntrance] = useState("А");
  const [apCount, setApCount] = useState(24);

  // DEMO “текуща такса” (пример)
  const dueDate = useMemo(() => {
    const d = new Date();
    d.setDate(10); // 10-то число (пример)
    d.setHours(23, 59, 59, 999);
    return d;
  }, []);

  // правим демо статус по апартамент номер (детерминирано)
  const getStatus = (apNumber) => {
    // 0..2 pattern
    const r = apNumber % 3;
    if (r === 0) return "paidOnTime";
    if (r === 1) return "overdue";
    return "unpaid";
  };

  const getStatusMeta = (s) => {
    if (s === "paidOnTime") {
      return {
        label: "Платено в срок",
        pill: "bg-emerald-100 text-emerald-800 border-emerald-200",
        tile: "bg-emerald-50 border-emerald-200 text-emerald-900",
      };
    }
    if (s === "overdue") {
      return {
        label: "Просрочено",
        pill: "bg-rose-100 text-rose-800 border-rose-200",
        tile: "bg-rose-50 border-rose-200 text-rose-900",
      };
    }
    return {
      label: "Неплатено",
      pill: "bg-amber-100 text-amber-800 border-amber-200",
      tile: "bg-amber-50 border-amber-200 text-amber-900",
    };
  };

  const apartments = useMemo(() => {
    return Array.from({ length: apCount }).map((_, i) => {
      const ap = i + 1;
      const status = getStatus(ap);
      const meta = getStatusMeta(status);

      // demo “кой е платил”
      const who =
        status === "paidOnTime"
          ? `Ап. ${ap} • Платил: Иван П.`
          : status === "overdue"
          ? `Ап. ${ap} • Последно плащане: преди 2 месеца`
          : `Ап. ${ap} • Няма плащане`;

      const due =
        `Срок: ${dueDate.toLocaleDateString("bg-BG")}`;

      return { ap, status, meta, who, due };
    });
  }, [apCount, dueDate]);

  // grid columns според броя апартаменти
  const cols = apCount <= 18 ? 6 : apCount <= 30 ? 6 : 8;

  return (
    <div className="rounded-3xl border border-sky-100 bg-white shadow-soft overflow-hidden">
      <div className="p-4 border-b border-sky-100 flex items-center justify-between gap-3">
        <div>
          <div className="text-xs text-slate-500">Визуализация (DEMO)</div>
          <div className="font-semibold text-slate-900">
            Вход {entrance} • всички апартаменти • hover за статус
          </div>
        </div>

        <div className="flex items-center gap-2">
          <span className="text-xs px-2 py-1 rounded-full bg-sky-100 text-sky-700 border border-sky-200">
            DEMO
          </span>
        </div>
      </div>

      <div className="p-4">
        {/* controls */}
        <div className="flex flex-col sm:flex-row sm:items-center gap-3 mb-4">
          <div className="flex items-center gap-2">
            <div className="text-xs text-slate-500">Избери вход:</div>
            <div className="flex gap-2">
              {["А", "Б", "В"].map((x) => (
                <button
                  key={x}
                  onClick={() => setEntrance(x)}
                  className={`text-xs px-3 py-1.5 rounded-2xl border transition ${
                    entrance === x
                      ? "bg-sky-600 text-white border-sky-600"
                      : "bg-white text-sky-700 border-sky-200 hover:bg-sky-50"
                  }`}
                >
                  {x}
                </button>
              ))}
            </div>
          </div>

          <div className="flex items-center gap-2 sm:ml-auto">
            <div className="text-xs text-slate-500">Апартаменти:</div>
            <select
              value={apCount}
              onChange={(e) => setApCount(Number(e.target.value))}
              className="text-xs border border-sky-200 rounded-2xl px-3 py-2 bg-white"
            >
              {[12, 18, 24, 30, 36].map((n) => (
                <option key={n} value={n}>
                  {n}
                </option>
              ))}
            </select>
          </div>
        </div>

        {/* grid */}
        <div
          className="grid gap-2"
          style={{ gridTemplateColumns: `repeat(${cols}, minmax(0, 1fr))` }}
        >
          {apartments.map((x) => (
            <div key={x.ap} className="relative group">
              <div
                className={`h-11 rounded-2xl border ${x.meta.tile} flex items-center justify-center text-xs font-black hover:scale-[1.03] transition cursor-default`}
              >
                {x.ap}
              </div>

              {/* tooltip */}
              <div className="pointer-events-none absolute left-1/2 -translate-x-1/2 -top-2 opacity-0 group-hover:opacity-100 group-hover:-translate-y-10 transition duration-150 z-10">
                <div className="w-56 rounded-2xl border border-sky-100 bg-white shadow-soft p-3">
                  <div className={`inline-flex text-[11px] px-2 py-1 rounded-full border ${x.meta.pill}`}>
                    {x.meta.label}
                  </div>
                  <div className="mt-2 text-xs text-slate-700 font-semibold">
                    Вход {entrance} • {x.who}
                  </div>
                  <div className="mt-1 text-[11px] text-slate-500">{x.due}</div>
                  <div className="mt-2 text-[11px] text-slate-500">
                    (Това е DEMO. После го връзваме към реалните плащания.)
                  </div>
                </div>
              </div>
            </div>
          ))}
        </div>

        {/* legend */}
        <div className="mt-4 flex flex-wrap items-center gap-3 text-xs text-slate-600">
          <div className="flex items-center gap-2">
            <span className="inline-block h-3 w-3 rounded bg-emerald-200 border border-emerald-300" />
            Платено в срок
          </div>
          <div className="flex items-center gap-2">
            <span className="inline-block h-3 w-3 rounded bg-rose-200 border border-rose-300" />
            Просрочено
          </div>
          <div className="flex items-center gap-2">
            <span className="inline-block h-3 w-3 rounded bg-amber-200 border border-amber-300" />
            Неплатено
          </div>
        </div>
      </div>
    </div>
  );
}

function PricingCard({ title, price, subtitle, items, primary, ctaText }) {
  return (
    <div
      className={`rounded-3xl border p-6 shadow-soft transition ${
        primary
          ? "border-sky-300 bg-gradient-to-b from-sky-50 to-white"
          : "border-slate-100 bg-white"
      }`}
    >
      <div className="flex items-start justify-between gap-3">
        <div>
          <div className="text-sm font-semibold text-sky-700">{title}</div>
          <div className="text-3xl font-black text-slate-900 mt-1">{price}</div>
          <div className="text-sm text-slate-600 mt-1 leading-relaxed">{subtitle}</div>
        </div>
        {primary && (
          <span className="text-xs px-3 py-1 rounded-full bg-sky-600 text-white">
            Най-добро
          </span>
        )}
      </div>

      <ul className="mt-5 space-y-2 text-sm text-slate-700">
        {items.map((x, i) => (
          <li key={i} className="flex gap-2">
            <span className="text-sky-600">✓</span>
            <span>{x}</span>
          </li>
        ))}
      </ul>

      <div className="mt-6">
        <Link
          to="/register"
          className={`inline-flex w-full items-center justify-center rounded-2xl px-5 py-3 text-sm font-semibold transition ${
            primary
              ? "bg-sky-600 text-white hover:bg-sky-700 shadow-soft"
              : "border border-sky-200 text-sky-700 hover:bg-sky-50"
          }`}
        >
          {ctaText || "Започни сега"}
        </Link>
      </div>
    </div>
  );
}
